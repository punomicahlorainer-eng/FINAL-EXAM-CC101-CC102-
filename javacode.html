Jumbled Letters Word Game


import java.util.Random; 
import java.util.Scanner; 

public class JumbledLettersWordGame { 
    static String scramble(String word, Random rand) { 

        char[] letters = word.toCharArray();
        String result = word; 

        while (result.equals(word)) { 
            for (int i = letters.length - 1; i > 0; i--) { 
                int j = rand.nextInt(i + 1);
                char temp = letters[i];
                letters[i] = letters[j];
                letters[j] = temp;
            }
            result = new String(letters);
        }
        return result;
    }

    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        Random rand = new Random();

        String[] words = {"JAVA", "PROGRAMMING", "COMPUTER", "TECHNOLOGY", "CODING"};
        
        int score = 0;
        int rounds = 3;
        
        System.out.println("=================================");
        System.out.println("   JUMBLED LETTERS WORD GAME");
        System.out.println("=================================");
        System.out.println("Welcome to Jumbled letters word game! Unscramble the word! You only have 3 tries in each word.");
        
        for (int r = 1; r <= rounds; r++) {
            
            String word = words[rand.nextInt(words.length)];
            
            String jumbled = scramble(word, rand);
            int points = 10;
            boolean solved = false;
            
            System.out.println("--- Word " + r + " of " + rounds + " ---");
            System.out.println("Jumbled: " + jumbled.toUpperCase());
             
            
            int tries = 3;
            while (tries > 0 && !solved) {
                System.out.print("Your answer (" + tries + " tries left): ");
                String answer = input.next().toUpperCase();
            
                if (answer.equals(word)) {
                    System.out.println("Correct! +10 points");
                    score += 10;
                    solved = true;
                } else {
                    tries--;
                    System.out.println("Wrong Word!");
                }
            }
            
            if (!solved) {
                System.out.println("Out of tries! The word is: " + word);
            }
            
            System.out.println("Total Score: " + score);
        }

            System.out.println("==========================================");
            System.out.println("Game is over! Your Final score is: " + score + " / " + (rounds * 10));
            System.out.println("Thanks for playing!");
            System.out.println("==========================================");
        
        input.close();
    }
}
